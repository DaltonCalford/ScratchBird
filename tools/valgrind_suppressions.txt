# GoogleTest framework (static allocations never freed)
{
   gtest_singleton_leak
   Memcheck:Leak
   match-leak-kinds: reachable
   ...
   fun:*testing*
}

{
   gtest_internal_init
   Memcheck:Leak
   match-leak-kinds: reachable
   ...
   obj:*/libgtest.so*
}
