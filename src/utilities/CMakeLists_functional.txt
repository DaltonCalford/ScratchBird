# CMakeLists.txt for Functional ScratchBird Utilities

cmake_minimum_required(VERSION 3.10)
project(ScratchBirdUtilities)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Find required packages
find_package(PkgConfig REQUIRED)

# Find ScratchBird client library
find_library(SCRATCHBIRD_LIB
    NAMES sbclient libsbclient
    PATHS /usr/lib /usr/local/lib
    PATH_SUFFIXES scratchbird
)

# Find readline library
find_library(READLINE_LIB readline)

# Include directories
include_directories(${CMAKE_CURRENT_SOURCE_DIR})
include_directories(${CMAKE_CURRENT_SOURCE_DIR}/../../include)
include_directories(${CMAKE_CURRENT_SOURCE_DIR}/../../include/firebird)

# Build the database framework
add_library(sb_database STATIC
    sb_database.cpp
    sb_database.h
)

# Link database framework with ScratchBird client library
target_link_libraries(sb_database ${SCRATCHBIRD_LIB})

# Build functional sb_isql
add_executable(sb_isql_functional
    sb_isql.cpp
)

target_link_libraries(sb_isql_functional 
    sb_database
    ${READLINE_LIB}
    ${SCRATCHBIRD_LIB}
)

# Build functional sb_gstat
add_executable(sb_gstat_functional
    sb_gstat_functional.cpp
)

target_link_libraries(sb_gstat_functional 
    sb_database
    ${SCRATCHBIRD_LIB}
)

# Build functional sb_gbak
add_executable(sb_gbak_functional
    sb_gbak_functional.cpp
)

target_link_libraries(sb_gbak_functional 
    sb_database
    ${SCRATCHBIRD_LIB}
)

# Build functional sb_gfix
add_executable(sb_gfix_functional
    sb_gfix_functional.cpp
)

target_link_libraries(sb_gfix_functional 
    sb_database
    ${SCRATCHBIRD_LIB}
)

# Build functional sb_gsec
add_executable(sb_gsec_functional
    sb_gsec_functional.cpp
)

target_link_libraries(sb_gsec_functional 
    sb_database
    ${SCRATCHBIRD_LIB}
)

# Compiler flags
target_compile_options(sb_database PRIVATE -Wall -Wextra)
target_compile_options(sb_isql_functional PRIVATE -Wall -Wextra)
target_compile_options(sb_gstat_functional PRIVATE -Wall -Wextra)
target_compile_options(sb_gbak_functional PRIVATE -Wall -Wextra)
target_compile_options(sb_gfix_functional PRIVATE -Wall -Wextra)
target_compile_options(sb_gsec_functional PRIVATE -Wall -Wextra)

# Install targets
install(TARGETS sb_isql_functional sb_gstat_functional sb_gbak_functional sb_gfix_functional sb_gsec_functional
    RUNTIME DESTINATION bin
)

# Create symlinks for standard names
install(CODE "execute_process(COMMAND ${CMAKE_COMMAND} -E create_symlink sb_isql_functional sb_isql)")
install(CODE "execute_process(COMMAND ${CMAKE_COMMAND} -E create_symlink sb_gstat_functional sb_gstat)")
install(CODE "execute_process(COMMAND ${CMAKE_COMMAND} -E create_symlink sb_gbak_functional sb_gbak)")
install(CODE "execute_process(COMMAND ${CMAKE_COMMAND} -E create_symlink sb_gfix_functional sb_gfix)")
install(CODE "execute_process(COMMAND ${CMAKE_COMMAND} -E create_symlink sb_gsec_functional sb_gsec)")
